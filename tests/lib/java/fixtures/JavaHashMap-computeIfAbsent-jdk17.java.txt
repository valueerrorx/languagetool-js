import java.util.*;
public class P { public static void main(String[] a){ Random r=new Random(7);
 for(int t=0;t<300;t++){ HashMap<Integer,Integer> m=new HashMap<>(); StringBuilder ops=new StringBuilder(); int n=r.nextInt(60)+1; int space=r.nextBoolean()?8:200; int mult=new int[]{1,16,64}[r.nextInt(3)];
  for(int i=0;i<n;i++){ int k=r.nextInt(space)*mult; int op=r.nextInt(3); ops.append(op).append(':').append(k).append(' ');
   if(op==0) m.computeIfAbsent(k,x->x+1); else if(op==1) m.put(k,k); else m.computeIfAbsent(k,x->null); }
  System.out.println(ops+"|"+m.keySet()); } } }
